cmake_minimum_required(VERSION 3.24)
project(ort_webgpu_native_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_INCLUDE_DIR "" CACHE PATH "ORT C/C++ API headers matching the runtime")
set(ORT_LIBRARY "" CACHE FILEPATH "onnxruntime import library")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
set(ORT_BUILD_DIR "" CACHE PATH "ORT build with the matching Dawn static libraries")
foreach(required ORT_INCLUDE_DIR ORT_LIBRARY JSON_INCLUDE_DIR)
  if(NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Set ${required} to an existing dependency")
  endif()
endforeach()
find_program(NODE_EXECUTABLE node REQUIRED)
set(SHADER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/greedy_shader.h")
add_custom_command(OUTPUT "${SHADER_HEADER}"
  COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/emit-shader.mjs" "${SHADER_HEADER}"
  DEPENDS emit-shader.mjs ../web/gpu-greedy.mjs VERBATIM)
add_executable(ort_webgpu_benchmark main.cpp "${SHADER_HEADER}")
target_include_directories(ort_webgpu_benchmark PRIVATE "${ORT_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_include_directories(ort_webgpu_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated"
  "${ORT_BUILD_DIR}/_deps/dawn-src/include" "${ORT_BUILD_DIR}/_deps/dawn-build/gen/include")
# Reuse the exact Dawn/Tint build that produced the native ORT runtime. Passing
# its procedure table and device to ORT keeps both users on the same device.
file(GLOB_RECURSE DAWN_LIBS CONFIGURE_DEPENDS "${ORT_BUILD_DIR}/_deps/dawn-build/*.lib")
file(GLOB_RECURSE ABSL_LIBS CONFIGURE_DEPENDS "${ORT_BUILD_DIR}/_deps/abseil_cpp-build/*.lib")
file(GLOB_RECURSE GLFW_LIBS CONFIGURE_DEPENDS "${ORT_BUILD_DIR}/_deps/dawn-build/*glfw*.lib")
list(REMOVE_ITEM DAWN_LIBS ${GLFW_LIBS})
target_link_libraries(ort_webgpu_benchmark PRIVATE ${DAWN_LIBS} ${ABSL_LIBS})
target_link_libraries(ort_webgpu_benchmark PRIVATE "${ORT_LIBRARY}")
if(WIN32)
  target_link_libraries(ort_webgpu_benchmark PRIVATE dxgi d3d11 d3d12 d3dcompiler dxguid user32 gdi32 advapi32 ole32 shell32 dbghelp bcrypt onecore)
endif()
if(MSVC)
  target_compile_options(ort_webgpu_benchmark PRIVATE /O2 /W4 /EHsc)
  target_compile_definitions(ort_webgpu_benchmark PRIVATE NOMINMAX)
endif()

# Separate diagnostic executable; never part of a timed benchmark run.
add_executable(native_logits_diagnostic diagnose-logits.cpp "${SHADER_HEADER}")
get_target_property(BENCHMARK_INCLUDES ort_webgpu_benchmark INCLUDE_DIRECTORIES)
get_target_property(BENCHMARK_LIBRARIES ort_webgpu_benchmark LINK_LIBRARIES)
target_include_directories(native_logits_diagnostic PRIVATE ${BENCHMARK_INCLUDES})
target_link_libraries(native_logits_diagnostic PRIVATE ${BENCHMARK_LIBRARIES})
if(MSVC)
  target_compile_options(native_logits_diagnostic PRIVATE /O2 /EHsc)
  target_compile_definitions(native_logits_diagnostic PRIVATE NOMINMAX)
endif()
